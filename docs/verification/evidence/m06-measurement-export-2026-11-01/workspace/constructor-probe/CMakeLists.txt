cmake_minimum_required(VERSION 3.25)
project(GenesisConstructorProbe LANGUAGES CXX)
if(NOT GENESIS_ROOT)
    message(FATAL_ERROR "GENESIS_ROOT is required")
endif()
add_executable(probe main.cpp)
file(GLOB libraries "${GENESIS_ROOT}/build/asan/RelWithDebInfo/*.lib")
file(GLOB runtime "${GENESIS_ROOT}/build/asan/RelWithDebInfo/*.dll")
add_executable(continue_probe continue.cpp)
foreach(target IN ITEMS probe continue_probe)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8 /fsanitize=address /Zi)
    target_include_directories(${target} PRIVATE "${GENESIS_ROOT}/src" "${GENESIS_ROOT}")
    target_link_libraries(${target} PRIVATE ${libraries})
    foreach(dll IN LISTS runtime)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${dll}" "$<TARGET_FILE_DIR:${target}>")
    endforeach()
endforeach()
